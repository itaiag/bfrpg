28  מחולל מבוכים

המחולל יוצר מפת מבוך אקראית לפי הכללים של בייסיק פנטזי. בחרי את גודל המבוך ואת הרכב החבורה ולחצי על הכפתור. אותו קוד מבוך, עם אותן הגדרות, יוצר תמיד את אותו מבוך, כך שאפשר לשמור את הקוד ולחזור אליו.

כל חדר מאוכלס לפי טבלת האכלוס שביצירת הרפתקאות: מפלצות מטבלאות המפגשים במבוך, מלכודות ואוצרות, וכולם מקושרים אל הכללים שלהם. המפלס הראשון מתאים לדרגת החבורה, וכל מפלס שמתחתיו קשה בדרגה אחת. מספר המפלצות בכל חדר מותאם לגודל החבורה: הגלגול שבכללים משמש לחבורה של ארבע דמויות.

למבוך יש גם סיפור: מי בנה אותו, מדוע ננטש, ואילו פלגים נאבקים עליו כיום. לכל פלג יש שטח משלו (צבוע במפה), מנהיג, מטרה ויחס אל הרפתקנים. המתאם לגלגול התגובה שמוצע לכל פלג הוא הצעה של המחולל ואינו חלק מהכללים.

במשחק סולו המחולל ממלא את תפקיד המנחה: הוא מסתיר את מה שהחבורה טרם גילתה, מגלגל הפתעה, תגובה, מורל ומפלצות נודדות, ועוקב אחר התורות ואחר מקור האור. את הקרבות עצמם מנהלים ליד השולחן לפי הכללים, ומדווחים למחולל כיצד הסתיימו.

import {createDungeon} from "./dungeon/index.js"
import {mount} from "./dungeon/render.js"
import {mountSolo} from "./dungeon/solo.js"
import {content} from "./dungeon/content-he.js"
import {t} from "./dungeon/strings-he.js"

dungeonData = FileAttachment("dungeon-data.json").json()
viewof dungeonConfig = Inputs.form({
  size: Inputs.select(["small", "medium", "large"], {label: t.form.size, value: "medium", format: (key) => t.form.sizes[key]}),
  levels: Inputs.range([1, 3], {label: t.form.levels, step: 1, value: 1}),
  partyCount: Inputs.range([1, 8], {label: t.form.partyCount, step: 1, value: 4}),
  partyLevel: Inputs.range([1, 10], {label: t.form.partyLevel, step: 1, value: 1}),
  classes: Inputs.checkbox(Object.keys(t.form.classNames), {label: t.form.classes, value: Object.keys(t.form.classNames), format: (key) => t.form.classNames[key]}),
  seed: Inputs.text({label: t.form.seed, placeholder: t.form.seedHint})
})

viewof dungeonView = Inputs.radio(["gm", "solo"], {label: t.view.label, value: "gm", format: (key) => t.view[key]})

viewof dungeonRequest = Inputs.button(t.form.generate)
dungeon = {
  dungeonRequest; // a new dungeon on every press, not on every change to the form
  const config = viewof dungeonConfig.value;
  return createDungeon({
    seed: config.seed.trim(),
    size: config.size,
    levels: config.levels,
    party: {count: config.partyCount, level: config.partyLevel, classes: config.classes}
  }, dungeonData, content);
}

dungeonLinks = ({...dungeonData.rules, stocking: dungeonData.stocking.link})

dungeonView === "solo" ? mountSolo(dungeon, t, dungeonLinks, dungeonData) : mount(dungeon, t, dungeonLinks)
